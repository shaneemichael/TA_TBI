% Single responsibility: canonical enumeration of the 5 preprocessing conditions.
% Change this file when a strategy is added/removed or its specification changes.
% \input from any document that needs to enumerate the strategies.

\begin{enumerate}
    \item \textbf{Keep} --- identity preprocessing; UTF-8 normalised to NFC. Baseline.
    \item \textbf{Naive strip} --- regex \texttt{(\textbackslash w+?)nya\textbackslash b} replaced with the captured group. Included as a deliberate harm-baseline; over-strips non-clitic words such as \id{punya}, \id{tanya}, \id{hanya}, and proper nouns / loanwords ending in \id{-nya} (\id{Kenya}, \id{Sonya}, \id{Tanya}).
    \item \textbf{Sastrawi clitic} --- Sastrawi's \texttt{RemoveInflectionalPossessivePronoun} visitor applied with an explicit Indonesian-dictionary guard; strips \id{-nya} only when the result is a known root, protecting non-clitic words. Note that adverbial \id{-nya} forms (\id{biasanya}, \id{karenanya}, \id{tahunya}) \emph{are} stripped because their roots (\id{biasa}, \id{karena}, \id{tahu}) are in Sastrawi's lexicon --- this is the correct dictionary-guarded behaviour and we treat it as a true clitic, not a false positive.
    \item \textbf{Sentinel} --- \id{-nya} replaced with a spaced sentinel token \texttt{<NYA>}. Diagnostic condition that removes lexical match while preserving the information that a clitic was present.
    \item \textbf{Rule-resolved} --- in-house single-pass \id{-nya} resolver: every \id{-nya} occurrence is detected via the same Sastrawi-guarded matcher used in Strategy~3, classified by function using surface-context heuristics, and anaphoric occurrences are substituted with the most recent salient capitalised noun phrase from the prior one to two sentences. The full passage corpus is rewritten in CPU minutes with no human annotation and no commercial API. This is a deliberately conservative surface heuristic, \emph{not} the IndoCoref multi-pass sieve system of Artari et al.~\cite{artari2021indocoref}; IndoCoref is unsuitable for our setting because it expects SACR-format pre-annotated mention spans rather than raw passage text, and adding an automatic mention detector would change the experimental contract (see \secref{sec:related}). Strategy~5 should therefore be read as a lower bound on what a full coreference system would deliver, not as a coreference baseline.
\end{enumerate}
