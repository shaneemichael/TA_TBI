% Change this file when a strategy is added/removed or its specification changes.
% \input from any document that needs to enumerate the strategies.

\begin{enumerate}
    \item \textbf{Keep}: identity preprocessing; UTF-8 normalised to NFC.
    \item \textbf{Naive strip}: regex \texttt{(\textbackslash w+?)nya\textbackslash b} replaced with the captured group; deliberately over-strips non-clitic words such as \id{punya}, \id{tanya}, \id{hanya}, \id{Kenya}, \id{Sonya}, and \id{Tanya}.
    \item \textbf{Sastrawi clitic}: Sastrawi's possessive-pronoun remover applied with an Indonesian-dictionary guard; strips \id{-nya} only when the result is a known root.
    \item \textbf{Sentinel}: \id{-nya} replaced with a spaced \texttt{<NYA>} token, preserving clitic presence while removing lexical match.
    \item \textbf{Rule-resolved}: an in-house single-pass surface heuristic that detects guarded \id{-nya}, classifies surface function, and substitutes anaphoric cases with a recent salient noun phrase. It is not the IndoCoref system~\cite{artari2021indocoref}.
\end{enumerate}
